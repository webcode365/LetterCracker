function sortLetters(s) { return s.toLowerCase().split('').sort().join(''); } // ================= MAIN SOLVER (SAFE + STABLE) ================= function solveAnagram() { const inputEl = document.getElementById('anagramInput'); const rawInput = inputEl.value.trim().replace(/\s/g,'').toLowerCase(); if (!rawInput || rawInput.length < 2) { showToast('Please enter at least 2 letters'); return; } const loading = document.getElementById('anagramLoading'); const results = document.getElementById('anagramResults'); loading.style.display = 'flex'; results.style.display = 'none'; // ⚠️ KEEP DELAY TO PREVENT UI LAYOUT BUGS setTimeout(() => { if (!window.WORD_DB || !WORD_DB.words) { loading.style.display = 'none'; showToast("Word database not loaded"); return; } const sortedInput = sortLetters(rawInput); // SAFE ORIGINAL MATCH LOGIC (NO BREAKAGE) const anagrams = WORD_DB.words.filter(word => word.length === rawInput.length && sortLetters(word) === sortedInput ); loading.style.display = 'none'; results.style.display = 'block'; document.getElementById('anagramMeta').innerHTML = `Found ${anagrams.length} anagram${anagrams.length !== 1 ? 's' : ''} for ${rawInput.toUpperCase()}`; const chipsEl = document.getElementById('anagramChips'); if (anagrams.length === 0) { chipsEl.innerHTML = `

No exact anagrams found. Try Word Unscrambler.

`; } else { chipsEl.innerHTML = anagrams.map(w => ` ${w.toUpperCase()} ${WORD_DB.getPoints(w)}pt ` ).join(''); } }, 200); } // ================= COPY FUNCTIONS ================= function copyAnagrams() { const chips = document.querySelectorAll('#anagramChips .word-chip'); const text = [...chips].map(c => c.textContent.replace(/\d+pt/g,'').trim() ).join('\n'); navigator.clipboard?.writeText(text).then(() => showToast('Copied!') ); } function copyWord(w) { navigator.clipboard?.writeText(w).then(() => showToast(`"${w}" copied!`) ); } // ================= TOAST ================= function showToast(msg) { const t = document.getElementById('toast'); t.textContent = msg; t.classList.add('show'); setTimeout(() => t.classList.remove('show'), 2000); } // ================= INPUT UI ================= const inp = document.getElementById('anagramInput'); const clr = document.getElementById('clearBtn'); inp.addEventListener('input', () => { inp.value = inp.value.toUpperCase().replace(/[^A-Z]/g,''); }); clr.addEventListener('click', () => { inp.value = ''; document.getElementById('anagramResults').style.display = 'none'; }); inp.addEventListener('keydown', e => { if (e.key === 'Enter') solveAnagram(); });
